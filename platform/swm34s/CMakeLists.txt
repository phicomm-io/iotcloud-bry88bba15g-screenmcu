set(SWM_SDK_DIR "" CACHE PATH "Unmodified official SWM341_Lib 2025-08-07")
option(SCREEN_RECOVERY_ONLY "Boot only internal-SRAM recovery service" OFF)
option(SCREEN_STARTUP_DIAGNOSTIC "Observe each business initialization stage for five seconds" OFF)
option(SCREEN_UART2_QUIET "Diagnostic image: suppress all PM/TVOC UART2 traffic" OFF)
set(SCREEN_BACKLIGHT_MAX_PERCENT "100" CACHE STRING "Diagnostic ceiling for backlight duty (0..100 percent)")
if(NOT SCREEN_BACKLIGHT_MAX_PERCENT MATCHES "^(0|[1-9][0-9]?|100)$")
 message(FATAL_ERROR "SCREEN_BACKLIGHT_MAX_PERCENT must be an integer from 0 to 100")
endif()
if(NOT EXISTS "${SWM_SDK_DIR}/SWM341_StdPeriph_Driver/CSL/CMSIS/DeviceSupport/SWM341.h")
 message(FATAL_ERROR "Set SWM_SDK_DIR to official SWM341_Lib")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/check_sdk.py ${SWM_SDK_DIR} RESULT_VARIABLE SDK_ERROR)
if(SDK_ERROR)
 message(FATAL_ERROR "Official SDK hashes do not match")
endif()
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/build_version.py ${CMAKE_BINARY_DIR}
 RESULT_VARIABLE VERSION_ERROR)
if(VERSION_ERROR)
 message(FATAL_ERROR "Screen Git identity could not be generated")
endif()
set(CSL ${SWM_SDK_DIR}/SWM341_StdPeriph_Driver/CSL)
file(READ ${CSL}/CMSIS/DeviceSupport/system_SWM341.c SYSTEM_SOURCE)
# The factory loader leaves PLL output enabled. Stop it before changing its
# dividers, allow the crystal to settle, and require the manual's two lock reads.
# A missing crystal/PLL still leaves UART recovery usable at the HRC clock.
string(REPLACE "\tSYS->PLLDIV &=" "\tdelay_3ms();\n\tSYS->PLLCR = (SYS->PLLCR | SYS_PLLCR_OFF_Msk) & ~SYS_PLLCR_OUTEN_Msk;\n\tSYS->PLLDIV &=" SYSTEM_SOURCE "${SYSTEM_SOURCE}")
string(REPLACE "while(SYS->PLLLOCK == 0);" "unsigned locked=0; for(volatile unsigned budget=2000000; budget && locked<2; --budget) { locked=SYS->PLLLOCK ? locked+1 : 0; } if(locked<2) { return; }" SYSTEM_SOURCE "${SYSTEM_SOURCE}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/system_SWM341.c "${SYSTEM_SOURCE}")
set(DRIVERS)
foreach(name port gpio uart lcd pwm adc flash timr)
 list(APPEND DRIVERS ${CSL}/SWM341_StdPeriph_Driver/SWM341_${name}.c)
endforeach()
add_executable(screen_firmware startup.S board.c rtc_bus.c application.c ${CMAKE_CURRENT_BINARY_DIR}/system_SWM341.c ${DRIVERS}
 ${CMAKE_SOURCE_DIR}/src/ui.c ${CMAKE_SOURCE_DIR}/src/startup_ui.c ${CMAKE_SOURCE_DIR}/src/responsive_font.c ${CMAKE_SOURCE_DIR}/src/fonts/metric_font.c ${CMAKE_SOURCE_DIR}/src/fonts/clock_font.c ${CMAKE_SOURCE_DIR}/src/fonts/zh16.c ${CMAKE_SOURCE_DIR}/src/fonts/zh32.c ${CMAKE_SOURCE_DIR}/src/fonts/clock72.c)
target_include_directories(screen_firmware PRIVATE . ${CMAKE_BINARY_DIR} ${CSL}/CMSIS/CoreSupport ${CSL}/CMSIS/DeviceSupport ${CSL}/SWM341_StdPeriph_Driver)
target_compile_definitions(screen_firmware PRIVATE SWM341)
target_compile_definitions(screen_firmware PRIVATE SCREEN_BACKLIGHT_MAX_PERCENT=${SCREEN_BACKLIGHT_MAX_PERCENT})
if(SCREEN_RECOVERY_ONLY)
 target_compile_definitions(screen_firmware PRIVATE SCREEN_RECOVERY_ONLY=1)
endif()
if(SCREEN_STARTUP_DIAGNOSTIC)
 target_compile_definitions(screen_firmware PRIVATE SCREEN_STARTUP_DIAGNOSTIC=1)
endif()
if(SCREEN_UART2_QUIET)
 target_compile_definitions(screen_firmware PRIVATE SCREEN_UART2_QUIET=1)
endif()
target_compile_options(screen_firmware PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(screen_firmware PRIVATE screen_core lvgl m)
target_link_options(screen_firmware PRIVATE --specs=nano.specs --specs=nosys.specs -Wl,-u,_printf_float -Wl,--gc-sections -T${CMAKE_SOURCE_DIR}/platform/application.ld -Wl,-Map=${CMAKE_BINARY_DIR}/screen_firmware.map -Wl,--print-memory-usage)
set_target_properties(screen_firmware PROPERTIES SUFFIX ".elf" RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
add_custom_command(TARGET screen_firmware POST_BUILD COMMAND ${CMAKE_OBJCOPY} -O binary --gap-fill 0xff --pad-to 0x7b000 $<TARGET_FILE:screen_firmware> ${CMAKE_BINARY_DIR}/screen_firmware.bin COMMAND ${CMAKE_SIZE} $<TARGET_FILE:screen_firmware>
 COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/package_firmware.py ${CMAKE_BINARY_DIR})
